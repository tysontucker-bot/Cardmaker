One-time setup: connect a Google Sheet
  1. Make a new Google Sheet in your school Google Drive and name it something like “Folder Jobs data.”
  2. In the sheet, go to Extensions › Apps Script. Delete what's there, then paste the script (Copy script button below).
  3. Click Deploy › New deployment. Pick the type Web app. Set Execute as: Me and Who has access: Anyone, then Deploy and allow access.
  4. Copy the Web app URL it gives you and paste it in the box above.
  5. Tap Test the link. A test row shows up in a new “Sessions” tab in your sheet. You can delete it.
// Folder Jobs data: receives sessions from the folder QR codes and adds them to this sheet.
// Paste into Extensions > Apps Script, then Deploy > New deployment > Web app
// (Execute as: Me, Who has access: Anyone).

var SESSION_HEADERS = ["Saved", "Date", "Student", "Folder", "Task", "Correct", "Incorrect", "Prompted", "Scored", "% on their own", "Item results", "Notes", "ID"];
var ITEM_HEADERS = ["Date", "Student", "Folder", "Task", "Item", "Result", "ID"];
var WORD = { "+": "Correct", "-": "Incorrect", "P": "Prompted" };

function doGet() {
  return ContentService.createTextOutput("Folder Jobs data is connected. Sessions will show up in your Google Sheet.");
}

function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    var d = JSON.parse(e.postData.contents);
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sessions = sheetFor_(ss, "Sessions", SESSION_HEADERS);
    var items = sheetFor_(ss, "Items", ITEM_HEADERS);
    var id = clean_(d.id);

    // the phone may resend a session it wasn't sure went through; skip repeats
    var last = sessions.getLastRow();
    if (id && last > 1) {
      var from = Math.max(2, last - 500);
      var ids = sessions.getRange(from, 13, last - from + 1, 1).getValues();
      for (var k = 0; k < ids.length; k++) if (String(ids[k][0]) === id) return text_("duplicate");
    }

    var res = (d.r || []).filter(function (r) { return r && WORD[r[1]]; });
    var c = 0, x = 0, p = 0;
    res.forEach(function (r) { if (r[1] === "+") c++; else if (r[1] === "-") x++; else p++; });
    var n = c + x + p;
    var date = parseDate_(d.date);
    var detail = res.map(function (r) { return r[0] + " " + (r[1] === "-" ? "−" : r[1]); }).join(", ");

    sessions.appendRow([new Date(), date, clean_(d.s), clean_(d.f), clean_(d.t), c, x, p, n, n ? c / n : "", clean_(detail), clean_(d.notes), id]);
    var row = sessions.getLastRow();
    sessions.getRange(row, 2).setNumberFormat("m/d/yyyy");
    sessions.getRange(row, 10).setNumberFormat("0%");

    if (res.length) {
      var rows = res.map(function (r) { return [date, clean_(d.s), clean_(d.f), clean_(d.t), clean_(r[0]), WORD[r[1]], id]; });
      var start = items.getLastRow() + 1;
      items.getRange(start, 1, rows.length, ITEM_HEADERS.length).setValues(rows);
      items.getRange(start, 1, rows.length, 1).setNumberFormat("m/d/yyyy");
    }
    return text_("ok");
  } catch (err) {
    return text_("error: " + err);
  } finally {
    lock.releaseLock();
  }
}

function sheetFor_(ss, name, headers) {
  var sh = ss.getSheetByName(name);
  if (!sh) {
    sh = ss.insertSheet(name);
    sh.appendRow(headers);
    sh.getRange(1, 1, 1, headers.length).setFontWeight("bold");
    sh.setFrozenRows(1);
  }
  return sh;
}

function parseDate_(s) {
  var m = String(s || "").match(/^(\d{4})-(\d{2})-(\d{2})$/);
  return m ? new Date(+m[1], +m[2] - 1, +m[3]) : new Date();
}

// text only, and never something the sheet would read as a formula
function clean_(v) {
  var s = String(v == null ? "" : v).slice(0, 2000);
  return /^[=+\-@]/.test(s) ? "'" + s : s;
}

function text_(s) {
  return ContentService.createTextOutput(s);
}
